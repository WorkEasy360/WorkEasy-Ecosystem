# --- Remote product integration (Phase 3) --------------------------------
#
# JOSE only: RFC 7515/7517/7519 signing, verification and JWK Set rendering
# for the OIDC identity tokens products verify offline. The authorization
# flow itself is not delegated to a library - it has to bind organization
# context, membership validity and entitlement, which is this platform's
# domain and not any library's. See docs/phase-3/03-sso-architecture.md s7
# for the evaluation against Authlib and django-oauth-toolkit.
#
# `cryptography` is PyJWT's asymmetric backend and is also used directly by
# platform/ecosystem/keys.py to generate and serialise key pairs. It is a
# direct dependency, not an accident of the extra, so it is pinned here.
#
# 2.15.1 (2026-10 audit): 2.14.0 let a deeply nested payload escape the
# unverified-decode peek as RecursionError instead of DecodeError
# (CVE-2026-101918 / GHSA-42vr-xj54-vc7v) - an unauthenticated 500 on /token.
PyJWT==2.15.1
cryptography==50.0.1

# --- Partner channel (Phase 7) ---------------------------------------------
#
# DNS TXT lookups for custom-domain ownership proof
# (commercial/partners/domains.py). Behind a swappable `TxtResolver`, with a
# timeout and a lifetime, so a slow or hostile resolver cannot hold a request.
# Pure Python; no other dependency.
# 2.9.0 fixes CVE-2026-114605 but needs Python 3.11+. The images and CI run
# 3.12; the 3.10 pin only keeps a local 3.10 venv installable.
dnspython==2.9.0; python_version >= "3.11"
dnspython==2.7.0; python_version < "3.11"

# --- Production runtime (Phase 12) -------------------------------------------
#
# The WSGI server the API image runs (deploy/gunicorn.conf.py). Before Phase 12
# no application server was pinned at all, so what served production would
# have been whatever someone installed by hand.
gunicorn==26.2.0
# The S3 adapter behind the FileStore port (platform/files/storage_s3.py).
# Production settings have always required W360_FILE_STORAGE_BACKEND=s3 and no
# adapter existed, so every file operation failed in production. The official
# SDK rather than a hand-written SigV4 signer: request signing is not code to
# own for four S3 calls.
boto3==1.43.105
